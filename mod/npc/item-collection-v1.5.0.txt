//===== Item Collection UI v1.5.0 ==========================================
//= Server side of the Item Collection window (client/index.js).
//= The collection is account-wide: deposited cards are kept in the mod
//= store's account scope under client.c.<card id>, which the window reads.
//=
//= Requests, from api.server.request('icui', ...):
//=   @icui <n> sync              -> ok|<era>|<zeny>
//=   @icui <n> bag <card>        -> ok|<count in bag>
//=   @icui <n> dep <card> <amt>  -> ok|<deposited>|<zeny>|<message>
//=   @icui <n> wd <card> <amt>   -> ok|<deposited>|<zeny>|<message>
//=   @icui <n> act <combo>       -> ok|<message>
//=   @icui <n> alert <card>      -> ok| and, if the card still needs copies, a chat line
//=   @icui <n> where <card>      -> ok|<mob>~<rate>~<level>~<name>|...
//=   @icui <n> bagall            -> ok|<card>:<amount>,...
//=   @icui <n> depm <card>:<amt>,...  -> ok|<zeny>|<refused>|<card>:<deposited>,...
//=   @icui <n> wdm <card>:<amt>,...   -> the same, withdrawing
//= Rates are the server's own, read live (1 = 0.01%). The last three can be
//= long, so their answers come in several @@reply parts.
//= Anything refused answers err|<message>. Level 0, so every check is here.
//=
//= Combos only look at what is deposited and use nothing up. Activating sets
//= client.a.<combo>; its bonus counts while that is set and the cards are there.
//=
//= Bonuses: one bonus_script per change, holding only the difference from
//= what this character already has. Flags 8 (gone at logout) + 512 (other
//= scripts' bonus_script_clear leaves it) + 2048 (equal deltas may repeat).
//= Every login starts from nothing and applies the whole total again.
//===========================================================================
-	script	ItemCollectionUI	-1,{
	end;

OnInit:
	bindatcmd "icui", strnpcinfo(3) + "::OnRequest", 0, 99;
	// Typing @collection opens the window too.
	bindatcmd "collection", strnpcinfo(3) + "::OnOpen", 0, 99;
	setarray .bonus$[0], "bonus bBaseAtk", "bonus bMatk", "bonus bMaxHP", "bonus bMaxSP", "bonus bDef", "bonus bMdef", "bonus bFlee", "bonus bHit", "bonus bStr", "bonus bAgi", "bonus bVit", "bonus bInt", "bonus bDex", "bonus bLuk", "bonus bFlee2", "bonus bCritical", "bonus bLongAtkRate", "bonus bAspdRate", "bonus bMaxHPrate", "bonus bMaxSPrate", "bonus2 bAddRace,RC_Demon", "bonus2 bAddRace,RC_Undead";
	setarray .kindA[0], 0, 2, 4, 6, 3, 5, 7, 2;
	setarray .valueA[0], 1, 10, 1, 1, 5, 1, 1, 5;
	setarray .kindB[0], 1, -1, -1, -1, -1, -1, -1, -1;
	setarray .valueB[0], 1, 0, 0, 0, 0, 0, 0, 0;
	setarray .mileAt[0], 10, 25, 50, 75, 100;
	setarray .mileGives$[0], "8:1,9:1,10:1,11:1,12:1,13:1", "18:2,19:2", "8:1,9:1,10:1,11:1,12:1,13:1,0:10,1:10", "18:3,19:3", "8:2,9:2,10:2,11:2,12:2,13:2,0:20,1:20";
	setarray .comboName$[1], "Poring Family", "Field Critters", "Restless Dead", "Orc Warband", "Sky Hunters", "Shadow Thieves", "Arcane Circle", "Fallen Knights", "Legends of Midgard", "Night Terrors";
	setarray .comboCards$[1], "4001,4004,4033,4196,4341", "4006,4002,4003,4009,4010", "4038,4025,4106,4110,4090", "4066,4071,4085,4256,4255", "4015,4325,4056,4040,4019", "4016,4020,4064,4042,4051", "4113,4190,4141,4171,4139", "4133,4136,4140,4320,4187", "4144,4142,4134,4131,4143", "4122,4116,4100,4212,4020";
	setarray .comboGives$[1], "13:5,14:3", "10:3,18:3", "11:2,19:5", "8:5,0:15", "12:5,16:5", "9:5,15:5", "11:5,1:15", "10:5,18:5", "8:2,9:2,10:2,11:2,12:2,13:2,17:5", "20:5,21:5";
	end;

OnOpen:
	dispbottom "@@event icui open";
	end;

OnPCLoginEvent:
	deletearray @icui_t;
	callsub S_Apply;
	end;

OnRequest:
	.@req = atoi(.@atcmd_parameters$[0]);
	if (.@req <= 0)
		end;
	.@op$ = .@atcmd_parameters$[1];
	if (getstrlen($@ICUI_era$) == 0) {
		.@out$ = "err|The card list did not load. Check the server log.";
	} else if (.@op$ == "sync") {
		.@out$ = "ok|" + $@ICUI_era$ + "|" + Zeny;
	} else if (.@op$ == "bag") {
		.@id = atoi(.@atcmd_parameters$[2]);
		// Not ?: because both sides of it always run, and countitem stops the script on a bad id.
		if ($@ICUI_tier[.@id])
			.@out$ = "ok|" + countitem(.@id);
		else
			.@out$ = "err|That is not a collectable card.";
	} else if (.@op$ == "dep" || .@op$ == "wd") {
		.@out$ = callsub(S_Move, .@op$, atoi(.@atcmd_parameters$[2]), atoi(.@atcmd_parameters$[3]));
	} else if (.@op$ == "where") {
		callsub S_Where, .@req, atoi(.@atcmd_parameters$[2]);
		end;
	} else if (.@op$ == "bagall") {
		callsub S_BagAll, .@req;
		end;
	} else if (.@op$ == "depm") {
		.@out$ = callsub(S_MoveMany, "dep", .@atcmd_parameters$[2]);
	} else if (.@op$ == "wdm") {
		.@out$ = callsub(S_MoveMany, "wd", .@atcmd_parameters$[2]);
	} else if (.@op$ == "alert") {
		.@id = atoi(.@atcmd_parameters$[2]);
		if ($@ICUI_tier[.@id] > 0 && getitemname(.@id) != "null") {
			.@need = callsub(S_Need, $@ICUI_tier[.@id]);
			.@have = modstore_get("account", "client.c." + .@id, 0);
			if (.@have < .@need)
				dispbottom "Item Collection: " + getitemname(.@id) + " " + .@have + " / " + .@need + " deposited, " + countitem(.@id) + " in your bag.";
		}
		.@out$ = "ok|";
	} else if (.@op$ == "act") {
		.@out$ = callsub(S_Activate, atoi(.@atcmd_parameters$[2]));
	} else {
		.@out$ = "err|Unknown request.";
	}
	dispbottom "@@reply " + .@req + " 1/1 " + .@out$;
	end;

// S_Where(request, card): each monster that drops the card, one reply part each.
S_Where:
	.@req = getarg(0);
	.@id = getarg(1);
	if ($@ICUI_tier[.@id] < 1) {
		dispbottom "@@reply " + .@req + " 1/1 err|That is not a collectable card.";
		return;
	}
	explode(.@mobs$, $@ICUI_mobs$[.@id], ",");
	.@parts = 0;
	for (.@i = 0; .@i < getarraysize(.@mobs$); .@i++) {
		.@mob = atoi(.@mobs$[.@i]);
		if (.@mob < 1 || getmonsterinfo(.@mob, MOB_LV) < 0 || !getmobdrops(.@mob))
			continue;
		for (.@d = 0; .@d < $@MobDrop_count; .@d++) {
			if ($@MobDrop_item[.@d] != .@id)
				continue;
			.@line$[.@parts] = .@mob + "~" + $@MobDrop_rate[.@d] + "~" + getmonsterinfo(.@mob, MOB_LV) + "~" + getmonsterinfo(.@mob, MOB_NAME) + "|";
			.@parts++;
			break;
		}
	}
	if (.@parts == 0) {
		dispbottom "@@reply " + .@req + " 1/1 ok|";
		return;
	}
	for (.@i = 0; .@i < .@parts; .@i++)
		dispbottom "@@reply " + .@req + " " + (.@i + 1) + "/" + .@parts + " " + (.@i ? "" : "ok|") + .@line$[.@i];
	return;

// S_BagAll(request): every collectable card in this character's bag.
S_BagAll:
	.@req = getarg(0);
	getinventorylist;
	.@parts = 0;
	.@inline = 0;
	.@line$[0] = "";
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		.@id = @inventorylist_id[.@i];
		if ($@ICUI_tier[.@id] < 1)
			continue;
		if (.@inline == 15) {
			.@parts++;
			.@inline = 0;
			.@line$[.@parts] = "";
		}
		.@line$[.@parts] += .@id + ":" + @inventorylist_amount[.@i] + ",";
		.@inline++;
	}
	.@parts++;
	for (.@i = 0; .@i < .@parts; .@i++)
		dispbottom "@@reply " + .@req + " " + (.@i + 1) + "/" + .@parts + " " + (.@i ? "" : "ok|") + .@line$[.@i];
	return;

// S_MoveMany("dep"|"wd", "card:amount,...") -> reply text. The bonus changes once, at the end.
S_MoveMany:
	.@how$ = getarg(0);
	explode(.@wants$, getarg(1), ",");
	.@done$ = "";
	.@refused = 0;
	for (.@i = 0; .@i < getarraysize(.@wants$) && .@i < 20; .@i++) {
		deletearray .@pair$;
		explode(.@pair$, .@wants$[.@i], ":");
		.@id = atoi(.@pair$[0]);
		deletearray .@result$;
		explode(.@result$, callsub(S_Move, .@how$, .@id, atoi(.@pair$[1]), 1), "|");
		if (.@result$[0] == "ok")
			.@done$ += .@id + ":" + .@result$[1] + ",";
		else
			.@refused++;
	}
	callsub S_Apply;
	return "ok|" + Zeny + "|" + .@refused + "|" + .@done$;

// S_Move("dep"|"wd", card id, amount{, quiet}) -> reply text. Quiet leaves the bonus to the caller.
S_Move:
	.@op$ = getarg(0);
	.@id = getarg(1);
	.@n = getarg(2);
	.@quiet = getarg(3, 0);
	.@tier = $@ICUI_tier[.@id];
	if (.@tier < 1 || getitemname(.@id) == "null")
		return "err|That is not a collectable card.";
	if (.@n < 1)
		return "err|Choose how many.";
	.@need = callsub(S_Need, .@tier);
	.@path$ = "client.c." + .@id;
	.@have = modstore_get("account", .@path$, 0);
	.@name$ = getitemname(.@id);
	if (.@op$ == "dep") {
		.@cap = (callfunc("F_ModSetting", "item-collection-ui", "store_extra", 0) ? 9999 : .@need);
		if (.@have >= .@cap)
			return "err|" + .@name$ + (.@cap == .@need ? " is already complete." : " is full.");
		.@bag = countitem(.@id);
		if (.@bag < 1)
			return "err|You have no " + .@name$ + " in your inventory.";
		.@n = min(.@n, .@cap - .@have, .@bag);
		// Dividing first, so a large amount can't overflow the multiplication.
		.@fee = callfunc("F_ModSetting", "item-collection-ui", "deposit_fee", 1000);
		if (.@fee && Zeny / .@fee < .@n)
			return "err|Depositing " + .@n + " needs " + callfunc("F_InsertComma", .@fee) + " Zeny each.";
		.@cost = .@n * .@fee;
		delitem .@id, .@n;
		if (!modstore_set("account", .@path$, .@have + .@n)) {
			getitem .@id, .@n;
			return "err|The collection is full. Nothing was taken.";
		}
		Zeny -= .@cost;
		.@have += .@n;
		.@msg$ = "Deposited " + .@n + " " + .@name$ + ".";
	} else {
		if (.@have < 1)
			return "err|No " + .@name$ + " is deposited.";
		.@n = min(.@n, .@have);
		.@fee = callfunc("F_ModSetting", "item-collection-ui", "withdraw_fee", 10000);
		if (.@fee && Zeny / .@fee < .@n)
			return "err|Withdrawing " + .@n + " needs " + callfunc("F_InsertComma", .@fee) + " Zeny each.";
		.@cost = .@n * .@fee;
		if (!checkweight(.@id, .@n))
			return "err|You cannot carry " + .@n + " more " + .@name$ + ".";
		if (!modstore_set("account", .@path$, .@have - .@n))
			return "err|The collection could not be saved. Nothing was returned.";
		Zeny -= .@cost;
		getitem .@id, .@n;
		.@have -= .@n;
		.@msg$ = "Withdrew " + .@n + " " + .@name$ + ".";
	}
	if (!.@quiet)
		callsub S_Apply;
	return "ok|" + .@have + "|" + Zeny + "|" + .@msg$;

// S_Activate(combo) -> reply text
S_Activate:
	.@c = getarg(0);
	if (.@c < 1 || .@c >= getarraysize(.comboCards$))
		return "err|That is not a combo.";
	if (!callsub(S_ComboMet, .@c))
		return "err|Deposit all the required cards first.";
	.@path$ = "client.a." + .@c;
	if (modstore_get("account", .@path$, 0))
		return "err|" + .comboName$[.@c] + " is already active.";
	if (!modstore_set("account", .@path$, 1))
		return "err|The collection could not be saved.";
	callsub S_Apply;
	return "ok|" + .comboName$[.@c] + " is now active.";

// S_ComboMet(combo) -> 1 when the account has deposited enough of every card
S_ComboMet:
	explode(.@ids$, .comboCards$[getarg(0)], ",");
	.@want = max(1, callfunc("F_ModSetting", "item-collection-ui", "combo_need", 3));
	for (.@i = 0; .@i < getarraysize(.@ids$); .@i++) {
		.@id = atoi(.@ids$[.@i]);
		.@tier = $@ICUI_tier[.@id];
		if (.@tier < 1)
			return 0;
		if (modstore_get("account", "client.c." + .@id, 0) < min(.@want, callsub(S_Need, .@tier)))
			return 0;
	}
	return 1;

// S_Need(tier) -> copies that complete a card of that tier
S_Need:
	switch (getarg(0)) {
	case 3: return max(1, callfunc("F_ModSetting", "item-collection-ui", "need_mvp", 1));
	case 2: return max(1, callfunc("F_ModSetting", "item-collection-ui", "need_boss", 3));
	}
	return max(1, callfunc("F_ModSetting", "item-collection-ui", "need_normal", 10));

// S_Apply: bring this character's collection bonus up to date.
S_Apply:
	if (callfunc("F_ModSetting", "item-collection-ui", "bonuses", 1)) {
		for (.@tier = 1; .@tier <= 3; .@tier++)
			.@need[.@tier] = callsub(S_Need, .@tier);
		freeloop(1);
		.@k = modstore_keys("account", "client.c", .@ids$);
		for (.@i = 0; .@i < .@k; .@i++) {
			.@id = atoi(.@ids$[.@i]);
			.@tier = $@ICUI_tier[.@id];
			if (.@tier < 1)
				continue;
			if (modstore_get("account", "client.c." + .@id, 0) < .@need[.@tier])
				continue;
			.@done++;
			.@c = $@ICUI_cat[.@id];
			.@t[.kindA[.@c]] += .valueA[.@c] * .@tier;
			if (.kindB[.@c] >= 0)
				.@t[.kindB[.@c]] += .valueB[.@c] * .@tier;
		}
		// Milestones: goals are a share of this era's cards, and pause if a card is withdrawn.
		for (.@m = 0; .@m < getarraysize(.mileAt) && $@ICUI_total; .@m++) {
			if (.@done * 100 < .mileAt[.@m] * $@ICUI_total)
				break;
			deletearray .@gives$;
			explode(.@gives$, .mileGives$[.@m], ",");
			for (.@g = 0; .@g < getarraysize(.@gives$); .@g++) {
				deletearray .@pair$;
				explode(.@pair$, .@gives$[.@g], ":");
				.@t[atoi(.@pair$[0])] += atoi(.@pair$[1]);
			}
		}
		for (.@c = 1; .@c < getarraysize(.comboCards$); .@c++) {
			if (!modstore_get("account", "client.a." + .@c, 0) || !callsub(S_ComboMet, .@c))
				continue;
			deletearray .@gives$;
			explode(.@gives$, .comboGives$[.@c], ",");
			for (.@g = 0; .@g < getarraysize(.@gives$); .@g++) {
				deletearray .@pair$;
				explode(.@pair$, .@gives$[.@g], ":");
				.@t[atoi(.@pair$[0])] += atoi(.@pair$[1]);
			}
		}
		freeloop(0);
	}
	.@s$ = "";
	for (.@j = 0; .@j < getarraysize(.bonus$); .@j++) {
		.@d = .@t[.@j] - @icui_t[.@j];
		if (.@d) {
			.@s$ += .bonus$[.@j] + "," + .@d + ";";
			@icui_t[.@j] = .@t[.@j];
		}
	}
	if (.@s$ != "")
		bonus_script "{" + .@s$ + "}", 2000000, 2568;
	return;
}
